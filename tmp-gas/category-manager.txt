// 類別管理：在記帳 APP 裡新增、改名、排序、刪除類別。
// 只會修改「類別設定」A 欄和「分類規則」的類別／狀態欄，不會改動「總帳」或各主體分頁裡已經記的帳。
const CATEGORY_PROTECTED_ = ['每月訂閱制'];          // 每月訂閱功能固定使用，不能改名或刪除
const CATEGORY_ALIAS_KEY_ = 'CATEGORY_RENAMES';       // 改名紀錄：讓內建關鍵字（例如「午餐」→ 餐飲）也對到新名稱
const CATEGORY_MAX_LEN_ = 20;

function catSheet_(ss){
  var s=ss.getSheetByName(CATEGORY_SHEET);
  if(!s)throw new Error('找不到「類別設定」分頁。');
  return s;
}
// 目前的類別（A 欄第 2 列起，略過空白），以及 A 欄原本用到第幾列
function catRead_(ss){
  var s=catSheet_(ss),last=s.getLastRow();
  var raw=last>1?s.getRange(2,1,last-1,1).getDisplayValues().map(function(r){return String(r[0]).trim()}):[];
  var list=[];raw.forEach(function(v){if(v&&list.indexOf(v)<0)list.push(v)});
  return {sheet:s,list:list,height:raw.length};
}
// 只寫回 A 欄；原本較長時，多出來的格子清空，不動其他欄
function catWrite_(info,list){
  var h=Math.max(info.height,list.length);
  if(!h)return;
  var values=[];for(var i=0;i<h;i++)values.push([i<list.length?list[i]:'']);
  info.sheet.getRange(2,1,h,1).setValues(values);
}
function catName_(v,label){
  var name=historyText_(String(v==null?'':v),label||'類別名稱',100).replace(/\s+/g,' ');
  if(!name)throw new Error('請輸入類別名稱。');
  if(name.length>CATEGORY_MAX_LEN_)throw new Error('類別名稱最多 '+CATEGORY_MAX_LEN_+' 個字。');
  if(name==='待確認')throw new Error('「待確認」是系統保留的名稱。');
  if(/[<>]/.test(name))throw new Error('類別名稱不能有 < 或 >。');
  return name;
}
function catAliases_(){
  try{return JSON.parse(PropertiesService.getScriptProperties().getProperty(CATEGORY_ALIAS_KEY_)||'{}')}catch(e){return {}}
}
// 內建規則判斷出的舊名稱，換成改名後的新名稱
function categoryAlias_(category){
  if(!category||category==='待確認')return category;
  var map=catAliases_(),seen={};
  while(map[category]&&!seen[category]){seen[category]=1;category=map[category]}
  return category;
}
function catCounts_(ss,list){
  var counts={},s=ss.getSheetByName(LEDGER_SHEET);
  list.forEach(function(c){counts[c]=0});
  if(s&&s.getLastRow()>1)s.getRange(2,7,s.getLastRow()-1,7).getDisplayValues().forEach(function(r){
    if(r[6]==='已作廢')return;
    if(counts.hasOwnProperty(r[0]))counts[r[0]]++;
  });
  return counts;
}
function catResult_(ss,list,message){
  return {ok:true,categories:list,protectedCategories:CATEGORY_PROTECTED_,counts:catCounts_(ss,list),message:message||''};
}
function catLocked_(fn){
  var lock=LockService.getScriptLock();lock.waitLock(10000);
  try{
    var ss=SpreadsheetApp.openById(ALLOWED_SPREADSHEET_ID);
    var result=fn(ss);
    SpreadsheetApp.flush();
    return result;
  }finally{lock.releaseLock()}
}

function getCategoryManager(){
  var ss=SpreadsheetApp.openById(ALLOWED_SPREADSHEET_ID);
  return catResult_(ss,catRead_(ss).list);
}
function addCategory(name){
  return catLocked_(function(ss){
    name=catName_(name);
    var info=catRead_(ss);
    if(info.list.indexOf(name)>=0)return catResult_(ss,info.list,'「'+name+'」已經在選單裡。');
    var list=info.list.concat([name]);
    catWrite_(info,list);
    return catResult_(ss,list,'已新增「'+name+'」，下次可以直接選。');
  });
}
function renameCategory(oldName,newName){
  return catLocked_(function(ss){
    oldName=String(oldName||'').trim();newName=catName_(newName,'新名稱');
    if(CATEGORY_PROTECTED_.indexOf(oldName)>=0)throw new Error('「'+oldName+'」是每月訂閱功能使用的類別，不能改名。');
    var info=catRead_(ss),i=info.list.indexOf(oldName);
    if(i<0)throw new Error('找不到「'+oldName+'」，請重新整理。');
    if(oldName===newName)return catResult_(ss,info.list,'名稱沒有變更。');
    if(info.list.indexOf(newName)>=0)throw new Error('已經有「'+newName+'」這個類別。');
    var list=info.list.slice();list[i]=newName;
    catWrite_(info,list);
    // 學過的關鍵字規則一起改到新名稱
    var rules=ss.getSheetByName(RULE_SHEET),changed=0;
    if(rules&&rules.getLastRow()>1){
      var col=rules.getRange(2,3,rules.getLastRow()-1,1),vals=col.getDisplayValues();
      vals=vals.map(function(r){if(r[0]===oldName){changed++;return [newName]}return [r[0]]});
      if(changed)col.setValues(vals);
    }
    var map=catAliases_();map[oldName]=newName;delete map[newName];
    PropertiesService.getScriptProperties().setProperty(CATEGORY_ALIAS_KEY_,JSON.stringify(map));
    return catResult_(ss,list,'已把「'+oldName+'」改名為「'+newName+'」。之後記帳會用新名稱；已經記的帳保留原本的名稱。');
  });
}
function moveCategory(name,direction){
  return catLocked_(function(ss){
    var info=catRead_(ss),i=info.list.indexOf(String(name||''));
    if(i<0)throw new Error('找不到這個類別，請重新整理。');
    var j=direction==='up'?i-1:i+1;
    if(j<0||j>=info.list.length)return catResult_(ss,info.list);
    var list=info.list.slice(),t=list[i];list[i]=list[j];list[j]=t;
    catWrite_(info,list);
    return catResult_(ss,list);
  });
}
function deleteCategory(name){
  return catLocked_(function(ss){
    name=String(name||'').trim();
    if(CATEGORY_PROTECTED_.indexOf(name)>=0)throw new Error('「'+name+'」是每月訂閱功能使用的類別，不能刪除。');
    var info=catRead_(ss),i=info.list.indexOf(name);
    if(i<0)throw new Error('找不到「'+name+'」，請重新整理。');
    if(info.list.length<=1)throw new Error('至少要保留一個類別。');
    var list=info.list.slice();list.splice(i,1);
    catWrite_(info,list);
    // 指到這個類別的學習規則先停用（保留在表上，之後可以手動改回）
    var rules=ss.getSheetByName(RULE_SHEET),off=0;
    if(rules&&rules.getLastRow()>1){
      var range=rules.getRange(2,3,rules.getLastRow()-1,3),vals=range.getDisplayValues();
      var status=vals.map(function(r){if(r[0]===name&&r[2]!=='停用'){off++;return ['停用']}return [r[2]]});
      if(off)rules.getRange(2,5,vals.length,1).setValues(status);
    }
    return catResult_(ss,list,'已從選單移除「'+name+'」。已經記的帳保留原本的類別'+(off?'；'+off+' 條自動分類規則已停用':'')+'。');
  });
}
